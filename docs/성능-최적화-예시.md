사용자 정의 성능 측정:

```typescript
// src/hooks/usePerformance.ts
'use client';

import { useEffect } from 'react';
import { apiLogger } from '@/lib/logger';

export function usePerformance(pageName: string) {
  useEffect(() => {
    // 페이지 로드 시간 측정
    if (typeof window !== 'undefined' && window.performance) {
      const navigation = performance.getEntriesByType('navigation')[0] as PerformanceNavigationTiming;

      if (navigation) {
        const metrics = {
          page: pageName,
          // DNS 조회 시간
          dnsTime: navigation.domainLookupEnd - navigation.domainLookupStart,
          // TCP 연결 시간
          tcpTime: navigation.connectEnd - navigation.connectStart,
          // 서버 응답 시간
          responseTime: navigation.responseEnd - navigation.requestStart,
          // DOM 로딩 시간
          domLoadTime: navigation.domContentLoadedEventEnd - navigation.domContentLoadedEventStart,
          // 총 로딩 시간
          totalTime: navigation.loadEventEnd - navigation.navigationStart,
        };

        apiLogger.info('페이지 성능 측정', metrics);
      }
    }
  }, [pageName]);
}
```

### 3. API 응답 시간 측정

```typescript
// src/lib/api-client.ts
import { createTimer } from '@/lib/logger';

export async function fetchWithMetrics<T>(
  url: string,
  options?: RequestInit
): Promise<T> {
  const timer = createTimer(`API 호출: ${url}`, 'API');

  try {
    const response = await fetch(url, options);

    if (!response.ok) {
      throw new Error(`HTTP ${response.status}: ${response.statusText}`);
    }

    const data = await response.json();
    timer.end();

    return data;
  } catch (error) {
    timer.end();
    throw error;
  }
}
```

**사용:**

```typescript
// 기존

// 측정 포함

// 콘솔에 "API 호출: /api/data 완료: 234ms" 출력
```

### 4. 컴포넌트 렌더링 시간 측정

```typescript
// src/components/PerformanceWrapper.tsx
'use client';

import { useEffect, useRef } from 'react';
import { metricsLogger } from '@/lib/logger';

interface Props {
  name: string;
  children: React.ReactNode;
}

export function PerformanceWrapper({ name, children }: Props) {
  const renderTime = useRef<number>(0);

  useEffect(() => {
    renderTime.current = performance.now();
  }, []);

  useEffect(() => {
    const duration = performance.now() - renderTime.current;

    if (duration > 100) {
      // 100ms 넘으면 경고
      metricsLogger.warn(`${name} 렌더링 느림`, {
        component: name,
        durationMs: duration,
      });
    } else {
      metricsLogger.debug(`${name} 렌더링`, {
        component: name,
        durationMs: duration,
      });
    }
  });

  return <>{children}</>;
}
```

### 7. 에러 경계 컴포넌트

```typescript
// src/components/ErrorBoundary.tsx
'use client';

import { Component, ReactNode } from 'react';
import { logger } from '@/lib/logger';

interface Props {
  children: ReactNode;
  fallback?: ReactNode;
}

interface State {
  hasError: boolean;
  error?: Error;
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: any) {
    // 에러 로깅
    logger.error('컴포넌트 에러', error, {
      componentStack: errorInfo.componentStack,
    });
  }

  render() {
    if (this.state.hasError) {
      return this.props.fallback || (
        <div className="p-8 text-center">
          <h2 className="text-2xl font-bold text-red-600 mb-4">
            문제가 발생했습니다
          </h2>
          <p className="text-gray-600 mb-4">
            {this.state.error?.message || '알 수 없는 오류'}
          </p>
          <button
            onClick={() => this.setState({ hasError: false })}
            className="px-4 py-2 bg-blue-600 text-white rounded hover:bg-blue-700"
          >
            다시 시도
          </button>
        </div>
      );
    }

    return this.props.children;
  }
}
```

### 8. 캐싱 유틸리티

```typescript
// src/lib/cache.ts
interface CacheEntry<T> {
  data: T;
  timestamp: number;
}

class SimpleCache<T> {
  private cache = new Map<string, CacheEntry<T>>();
  private ttl: number; // Time to live (ms)

  constructor(ttlMinutes: number = 5) {
    this.ttl = ttlMinutes * 60 * 1000;
  }

  get(key: string): T | null {
    const entry = this.cache.get(key);

    if (!entry) return null;

    // 만료 확인
    if (Date.now() - entry.timestamp > this.ttl) {
      this.cache.delete(key);
      return null;
    }

    return entry.data;
  }

  set(key: string, data: T): void {
    this.cache.set(key, {
      data,
      timestamp: Date.now(),
    });
  }

  clear(): void {
    this.cache.clear();
  }
}

// RAG 결과 캐싱 예시
export const ragCache = new SimpleCache<any>(10); // 10분 캐싱
```

### 9. 디바운스 Hook

```typescript
// src/hooks/useDebounce.ts
'use client';

import { useState, useEffect } from 'react';

export function useDebounce<T>(value: T, delay: number = 500): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
}
```

**사용 (검색 입력):**

```typescript
// src/components/SearchInput.tsx

    />
  );
}

// src/app/api/agent-orchestrator/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { apiLogger, createTimer, measureAsync } from '@/lib/logger';
import { ragCache } from '@/lib/cache';

export const revalidate = 60; // 1분 캐싱

export async function POST(req: NextRequest) {
  const requestId = Math.random().toString(36).substr(2, 9);
  const overallTimer = createTimer('전체 처리', 'API');

  apiLogger.info('요청 수신', { requestId });

  try {
    // 요청 파싱
    const { userInput, followupAnswers } = await measureAsync(
      '요청 파싱',
      () => req.json(),
      'API'
    );

    // 캐시 확인
    const cacheKey = `automation:${userInput}`;
    const cached = ragCache.get(cacheKey);

    if (cached) {
      apiLogger.info('캐시 히트', { requestId });
      overallTimer.end();
      return NextResponse.json(cached);
    }

    // AI 처리
    const result = await measureAsync(
      '3단계 자동화 생성',
      () => generate3StepAutomation(userInput, followupAnswers),
      'AI'
    );

    // 캐싱
    ragCache.set(cacheKey, result);

    overallTimer.end();
    apiLogger.info('처리 완료', {
      requestId,
      cardsCount: result.cards.length,
    });

    return NextResponse.json(result);

  } catch (error) {
    overallTimer.end();
    apiLogger.error('처리 실패', error as Error, { requestId });

    return NextResponse.json(
      { error: '자동화 생성 실패' },
      { status: 500 }
    );
  }
}
```
